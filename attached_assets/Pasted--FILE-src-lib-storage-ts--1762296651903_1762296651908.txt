export const TERMS_VERSION = '2025-11-01'; // bump when Terms/Privacy change
export const LS_TERMS = 'cc_terms';
export const LS_ONB = 'cc_onboarding';
export const LS_PROFILE = 'cc_profile';

type TermsState = { accepted: boolean; version: string; ts: number };
type OnbState = { completed: boolean; ts: number; version: string };

export function getTerms(): TermsState | null {
  const raw = localStorage.getItem(LS_TERMS);
  return raw ? JSON.parse(raw) : null;
}
export function setTerms(accepted: boolean): void {
  const payload: TermsState = { accepted, version: TERMS_VERSION, ts: Date.now() };
  localStorage.setItem(LS_TERMS, JSON.stringify(payload));
}

export function getOnboarding(): OnbState | null {
  const raw = localStorage.getItem(LS_ONB);
  return raw ? JSON.parse(raw) : null;
}
export function setOnboardingCompleted(): void {
  const payload: OnbState = { completed: true, version: 'v1', ts: Date.now() };
  localStorage.setItem(LS_ONB, JSON.stringify(payload));
}

import React from 'react';
import { Redirect } from 'wouter';
import { useAuth } from '../auth/AuthContext';
import { getTerms, TERMS_VERSION, getOnboarding } from '../lib/storage';

export const RequireAuth: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const { isAuthenticated, isLoading } = useAuth();
  if (isLoading) return null;
  if (!isAuthenticated) return <Redirect to="/" />;
  return <>{children}</>;
};

export const RequireTerms: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const state = getTerms();
  const valid = !!state?.accepted && state.version === TERMS_VERSION;
  if (!valid) return <Redirect to="/terms" />;
  return <>{children}</>;
};

export const RequireOnboarding: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const onb = getOnboarding();
  if (!onb?.completed) return <Redirect to="/onboarding" />;
  return <>{children}</>;
};

// ──────────────────────────────────────────────────────────────
import React from 'react';
import { useLocation } from 'wouter';
import { setOnboardingCompleted } from '../../lib/storage';

type Step = 'welcome' | 'permissions' | 'location' | 'profile' | 'tutorial' | 'complete';

export default function Onboarding() {
  const [, setLoc] = useLocation();
  const [step, setStep] = React.useState<Step>('welcome');

  const next = () => {
    setStep((s) => {
      const order: Step[] = ['welcome','permissions','location','profile','tutorial','complete'];
      const i = order.indexOf(s);
      return order[Math.min(i + 1, order.length - 1)];
    });
  };
  const skip = () => next();

  React.useEffect(() => {
    if (step === 'complete') {
      setOnboardingCompleted();
      setLoc('/home');
    }
  }, [step, setLoc]);

  return (
    <div className="max-w-screen-sm mx-auto p-6">
      {step === 'welcome' && (
        <Screen title="Vanakkam! 👋"
          body="CartCircle helps you discover nearby ONDC stores and deals."
          next={next} skip={skip} />
      )}
      {step === 'permissions' && (
        <Screen title="Permissions"
          body="We’ll request location (nearby stores) and notifications (order status). You can skip."
          next={next} skip={skip} />
      )}
      {step === 'location' && (
        <Screen title="Set your location"
          body="Auto-detect or choose manually. (Plug in your FullscreenLocationModal V3 here.)"
          next={next} skip={skip} />
      )}
      {step === 'profile' && (
        <Screen title="Profile (optional)"
          body="First name, phone*, interests to personalize offers."
          next={next} skip={skip} />
      )}
      {step === 'tutorial' && (
        <Screen title="Quick tour"
          body="Learn discovery, cart, and order tracking in 30 seconds."
          next={next} skip={skip} />
      )}
      {step === 'complete' && null}
    </div>
  );
}

function Screen({
  title, body, next, skip,
}: { title: string; body: string; next: () => void; skip: () => void }) {
  return (
    <div>
      <h1 className="text-xl font-semibold">{title}</h1>
      <p className="mt-3 opacity-80">{body}</p>
      <div className="mt-6 flex gap-3">
        <button onClick={next} className="px-5 py-3 rounded-md bg-black text-white">Continue</button>
        <button onClick={skip} className="px-5 py-3 rounded-md border">Skip</button>
      </div>
    </div>
  );
}

// FILE: src/App.tsx

import React from 'react';
import { Route, Switch } from 'wouter';
import { AuthProvider } from './auth/AuthContext';
import { RequireAuth, RequireTerms, RequireOnboarding } from './routing/Guards';
import Landing from './pages/Landing';
import Terms from './pages/Terms';
import Onboarding from './pages/onboarding/Onboarding';
import Home from './pages/Home';
import Search from './pages/Search';
import Cart from './pages/Cart';
import Orders from './pages/Orders';
import Profile from './pages/Profile';
import Support from './pages/Support';
import Checkout from './pages/Checkout';
import ProductDetail from './pages/ProductDetail';

export default function App() {
  return (
    <AuthProvider>
      <Switch>
        <Route path="/" component={Landing} />
        <Route path="/terms" component={Terms} />
        <Route path="/onboarding" component={Onboarding} />

        <Route path="/home">
          <RequireAuth>
            <RequireTerms>
              <RequireOnboarding>
                <Home />
              </RequireOnboarding>
            </RequireTerms>
          </RequireAuth>
        </Route>

        <Route path="/search">
          <RequireAuth><RequireTerms><RequireOnboarding><Search /></RequireOnboarding></RequireTerms></RequireAuth>
        </Route>
        <Route path="/cart">
          <RequireAuth><RequireTerms><RequireOnboarding><Cart /></RequireOnboarding></RequireTerms></RequireAuth>
        </Route>
        <Route path="/orders">
          <RequireAuth><RequireTerms><RequireOnboarding><Orders /></RequireOnboarding></RequireTerms></RequireAuth>
        </Route>
        <Route path="/profile">
          <RequireAuth><RequireTerms><RequireOnboarding><Profile /></RequireOnboarding></RequireTerms></RequireAuth>
        </Route>
        <Route path="/support">
          <RequireAuth><RequireTerms><RequireOnboarding><Support /></RequireOnboarding></RequireTerms></RequireAuth>
        </Route>
        <Route path="/checkout">
          <RequireAuth><RequireTerms><RequireOnboarding><Checkout /></RequireOnboarding></RequireTerms></RequireAuth>
        </Route>

        <Route path="/product/:id">
          {(params) => (
            <RequireAuth><RequireTerms><RequireOnboarding><ProductDetail id={params.id} /></RequireOnboarding></RequireTerms></RequireAuth>
          )}
        </Route>

        <Route>404</Route>
      </Switch>
    </AuthProvider>
  );
}
